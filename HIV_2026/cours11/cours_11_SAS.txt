    select * /* avec * nous indiquon que bous voulons afficher toutes les colonnes*/
    from SASHELP.BASEBALL (OBS=10); /* on pourrait mettre aussi (OBS=10) afin 
    d'indiquer le nombre d'observations à afficher*/

* GROUP BY ; 

Proc sql; 
title 'nombre de homerun par équipe en 1986';
SELECT team, sum(nhome) as sum_homerun
FROM SASHELP.BASEBALL 
GROUP BY team;
quit; 

Proc sql; 
title 'nombre de homerun par équipe/position en 1986';
SELECT team, position, sum(nhome) as sum_homerun
FROM SASHELP.BASEBALL 
GROUP BY team, position;
quit;

PROC SQL ;
title "Statistique sommaire par équipe et position"; 
select team, 
		count(team) as count_player,
		nmiss(salary) as n_missing_salary,
		avg(salary) as mean_salary
from SASHELP.BASEBALL
group by team;
quit;


*HAVING; 

Proc sql; 
title 'nombre de homerun par équipe/position en 1986';
SELECT team, position, sum(nhome) as sum_homerun, avg(nhome) as mean_nhome
FROM SASHELP.BASEBALL 
WHERE natbat > 300
GROUP BY team, position
HAVING position like "%B";
quit; 


Proc sql; 
title 'nombre de homerun par équipe/position en 1986';
SELECT team, position, sum(nhome) as sum_homerun, avg(nhome) as mean_nhome
FROM SASHELP.BASEBALL 
where position like "%B"
GROUP BY team, position; 
quit;

*Creer variable catégorique ; 

PROC SQL ;
SELECT name, 
		position, 
		team, 
		nhome, 
		case 
		when nhome > 30 then 'Tres bon'
		when nhome >10 then 'Bon'
		else 'Pas bon'
		end as nhome_categorie
		
FROM SASHELP.BASEBALL;
quit;
